const corsHeaders: Record<string, string> = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

class AiError extends Error {
  status: number; code: string;
  constructor(code: string, message: string, status: number) {
    super(message); this.name = 'AiError'; this.code = code; this.status = status;
  }
}
const unauthorized = (m = 'Não autenticado.') => new AiError('unauthorized', m, 401);
const forbidden = (m = 'Acesso negado.') => new AiError('forbidden', m, 403);
const badRequest = (m = 'Requisição inválida.') => new AiError('bad_request', m, 400);
const rateLimited = (m = 'Muitas requisições. Tente novamente em instantes.') => new AiError('rate_limited', m, 429);
const aiFailure = (m = 'Não foi possível gerar o conteúdo agora. Tente novamente.') => new AiError('ai_failure', m, 502);
function jsonResponse(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, 'Content-Type': 'application/json' } });
}
function errorResponse(err: unknown): Response {
  if (err instanceof AiError) return jsonResponse({ error: { code: err.code, message: err.message } }, err.status);
  return jsonResponse({ error: { code: 'internal_error', message: 'Erro interno. Tente novamente.' } }, 500);
}

// Schema de SAÍDA (Zod)
const analyzeAssessmentOutputSchema = z.object({
  analysis: z.string().min(1),
  recommendations: z
    .array(
      z.object({
        category: z.string().min(1),
        text: z.string().min(1),
        priority: z.enum(['alta', 'media', 'baixa']),
      }),
    )
    .min(1),
  disclaimer: z.string().min(1),
});

// Structured Outputs NÃO aceita `minItems`/`minimum`/etc. no schema cru (400).
// Só type/required/enum/properties/additionalProperties. Os limites reais
// (lista não vazia, campos obrigatórios) são garantidos pelo Zod de saída
// (`analyzeAssessmentOutputSchema`) após a resposta.
const TOOL_INPUT_SCHEMA = {
  type: 'object',
  additionalProperties: false,
  required: ['analysis', 'recommendations', 'disclaimer'],
  properties: {
    analysis: { type: 'string' },
    recommendations: {
      type: 'array',
      items: {
        type: 'object',
        additionalProperties: false,
        required: ['category', 'text', 'priority'],
        properties: {
          category: { type: 'string' },
          text: { type: 'string' },
          priority: { type: 'string', enum: ['alta', 'media', 'baixa'] },
        },
      },
    },
    disclaimer: { type: 'string' },
  },
} as const;

const num = (v: unknown): number | null =>
  v == null ? null : typeof v === 'number' ? v : Number.isFinite(Number(v)) ? Number(v) : null;

// Transforma uma linha de physical_assessments num objeto legível em pt-BR,
// omitindo campos nulos. É isto (não o assessment_id) que a IA recebe.
function formatarAvaliacao(row: any): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  if (row.assessed_at) out.data_avaliacao = row.assessed_at;
  if (num(row.age) != null) out.idade = num(row.age);
  if (row.gender) out.genero = GENERO_PT[row.gender] ?? row.gender;
  if (row.protocol) out.protocolo = row.protocol;
  if (row.goal) out.objetivo = OBJETIVO_PT[row.goal] ?? row.goal;

  const comp: Record<string, number> = {};
  if (num(row.weight_kg) != null) comp.peso_kg = num(row.weight_kg)!;
  if (num(row.height_cm) != null) comp.altura_cm = num(row.height_cm)!;
  if (num(row.bmi) != null) comp.imc = num(row.bmi)!;
  if (num(row.body_fat_pct) != null) comp.percentual_gordura = num(row.body_fat_pct)!;
  if (num(row.fat_mass_kg) != null) comp.massa_gorda_kg = num(row.fat_mass_kg)!;
  if (num(row.lean_mass_kg) != null) comp.massa_magra_kg = num(row.lean_mass_kg)!;
  if (num(row.muscle_mass_kg) != null) comp.massa_muscular_kg = num(row.muscle_mass_kg)!;
  // `inbody_score` descontinuado: não é mais incluído na composição enviada à IA.
  if (Object.keys(comp).length) out.composicao_corporal = comp;

  const circ: Record<string, number> = {};
  for (const k of Object.keys(CIRC_PT)) {
    const v = num(row[k]);
    if (v != null) circ[CIRC_PT[k]] = v;
  }
  if (Object.keys(circ).length) out.circunferencias_cm = circ;

  const dobras: Record<string, number> = {};
  for (const k of ['skf_triceps', 'skf_subscapular', 'skf_chest', 'skf_midaxillary', 'skf_suprailiac', 'skf_abdominal', 'skf_thigh']) {
    const v = num(row[k]);
    if (v != null) dobras[DOBRA_PT[k]] = v;
  }
  const measurements = row.measurements && typeof row.measurements === 'object' ? row.measurements : {};
  if (num(measurements.skf_biceps) != null) dobras[DOBRA_PT.skf_biceps] = num(measurements.skf_biceps)!;
  if (Object.keys(dobras).length) out.dobras_cutaneas_mm = dobras;

  const anamnese =
    row.anamnesis && typeof row.anamnesis === 'object' && typeof row.anamnesis.texto === 'string'
      ? row.anamnesis.texto
      : null;
  if (anamnese) out.anamnese = anamnese;

  return out;
}

const JOB_TYPE = 'analyze_assessment';
const TOOL_NAME = 'analyze_assessment';
// Análise + recomendações longas podem passar de 4096 tokens e truncar o JSON
// (stop_reason=max_tokens). 8192 dá folga.
const MAX_OUTPUT_TOKENS = 8192;
const TIMEOUT_MS = 60_000;
const RATE_LIMIT_MAX = 10;
const RATE_LIMIT_WINDOW_MS = 60 * 60 * 1000;

const inputSchema = z.object({
  assessment_id: z.string().uuid(),
  previous_assessment_id: z.string().uuid().optional(),
});

function pricingForModel(model: string | null) {
  const name = (model ?? '').toLowerCase();
  if (name.includes('opus')) return { inputPerMTok: 4, outputPerMTok: 20 };
  if (name.includes('haiku')) return { inputPerMTok: 0.1, outputPerMTok: 0.5 };
  return { inputPerMTok: 2, outputPerMTok: 10 };
}
function estimateCost(model: string | null, input: number | null, output: number | null): number {
  const p = pricingForModel(model);
  return Math.max(0, (Math.max(0, input ?? 0) / 1e6) * p.inputPerMTok + (Math.max(0, output ?? 0) / 1e6) * p.outputPerMTok);
}

const ANTHROPIC_URL = 'https://api.anthropic.com/v1/messages';
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

async function callAnthropic(system: string, userContent: string) {
  const apiKey = Deno.env.get('ANTHROPIC_API_KEY');
  const model = Deno.env.get('ANTHROPIC_MODEL');
  if (!apiKey) throw aiFailure('ANTHROPIC_API_KEY não configurado.');
  if (!model) throw aiFailure('ANTHROPIC_MODEL não configurado.');
  // Structured Outputs nativo (output_config.format) — Sonnet 5.5 não aceita
  // tool_choice forçado. A resposta vem como JSON no bloco de texto.
  const body = JSON.stringify({
    model, max_tokens: MAX_OUTPUT_TOKENS, system,
    messages: [{ role: 'user', content: userContent }],
    output_config: { format: { type: 'json_schema', schema: TOOL_INPUT_SCHEMA } },
  });
  const maxRetries = 2;
  let lastErr: unknown;
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), TIMEOUT_MS);
    try {
      const res = await fetch(ANTHROPIC_URL, {
        method: 'POST',
        headers: { 'x-api-key': apiKey, 'anthropic-version': '2023-06-01', 'content-type': 'application/json' },
        body, signal: controller.signal,
      });
      if (!res.ok) {
        let errBody = '';
        try { errBody = await res.text(); } catch { errBody = '(sem corpo)'; }
        console.error(`[analyze-assessment] Anthropic HTTP ${res.status}: ${errBody}`);
        if ((res.status === 429 || res.status >= 500) && attempt < maxRetries) {
          lastErr = new Error(`status ${res.status}`); await sleep(250 * (attempt + 1)); continue;
        }
        throw aiFailure();
      }
      const json = await res.json();
      return { response: json, inputTokens: json?.usage?.input_tokens ?? null, outputTokens: json?.usage?.output_tokens ?? null, model: json?.model ?? model };
    } catch (err) {
      lastErr = err;
      if (attempt < maxRetries) { await sleep(250 * (attempt + 1)); continue; }
    } finally { clearTimeout(timer); }
  }
  void lastErr; throw aiFailure();
}

// Extrai o JSON estruturado (output_config.format): vem como texto JSON.
function extractStructuredOutput(response: any): unknown {
  const blocks = response?.content ?? [];
  const texto = blocks
    .filter((b: any) => b?.type === 'text' && typeof b?.text === 'string')
    .map((b: any) => b.text)
    .join('');
  if (!texto) {
    console.error('[analyze-assessment] Sem bloco de texto. stop_reason=' + String(response?.stop_reason));
    return undefined;
  }
  try {
    return JSON.parse(texto);
  } catch {
    if (response?.stop_reason === 'max_tokens') {
      console.error('[analyze-assessment] Resposta truncada por max_tokens — aumente MAX_OUTPUT_TOKENS.');
    } else {
      console.error('[analyze-assessment] Texto não é JSON válido: ' + texto.slice(0, 500));
    }
    return undefined;
  }
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: corsHeaders });

  const supabaseUrl = Deno.env.get('SUPABASE_URL')!;
  const anonKey = Deno.env.get('SUPABASE_ANON_KEY')!;
  const serviceRoleKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!;
  const admin = createClient(supabaseUrl, serviceRoleKey, { auth: { persistSession: false, autoRefreshToken: false } });

  try {
    if (req.method !== 'POST') throw badRequest('Método não suportado.');

    const authHeader = req.headers.get('Authorization');
    const token = (authHeader ?? '').replace(/^Bearer\s+/i, '').trim();
    if (!token) throw unauthorized();
    const anon = createClient(supabaseUrl, anonKey, {
      auth: { persistSession: false, autoRefreshToken: false },
      global: { headers: { Authorization: `Bearer ${token}` } },
    });
    const { data: userData, error: userErr } = await anon.auth.getUser(token);
    if (userErr || !userData?.user) throw unauthorized();
    const userId = userData.user.id;

    const { data: profile } = await admin.from('profiles').select('tenant_id, role').eq('user_id', userId).maybeSingle();
    if (!profile) throw forbidden('Perfil não encontrado.');
    if (profile.role !== 'trainer') throw forbidden('Apenas o professor pode usar os recursos de IA.');
    const tenantId = profile.tenant_id as string;

    let rawBody: unknown = {};
    try { rawBody = await req.json(); } catch { rawBody = {}; }
    const parsedInput = inputSchema.safeParse(rawBody);
    if (!parsedInput.success) throw badRequest('Entrada inválida para análise de avaliação.');
    const input = parsedInput.data;

    // consentimento removido por decisão de escopo; reintroduzir se LGPD voltar.
    // (Sem o consentimento, não é preciso resolver o aluno dono da avaliação
    // aqui; a validação de posse/tenant fica por conta da própria geração.)

    // Carrega a avaliação (service role; valida tenant). SEM isto a IA recebia
    // só o assessment_id e respondia "sem dados para analisar". Feito ANTES de
    // registrar o pending para não deixar geração órfã quando a avaliação não
    // existe ou é de outro tenant.
    const { data: avaliacao, error: erroAval } = await admin
      .from('physical_assessments')
      .select(ASSESSMENT_COLUMNS)
      .eq('id', input.assessment_id)
      .maybeSingle();
    if (erroAval || !avaliacao) throw badRequest('Avaliação não encontrada.');
    if (avaliacao.tenant_id !== tenantId) throw forbidden('Avaliação de outro tenant.');

    const dados: Record<string, unknown> = { avaliacao_atual: formatarAvaliacao(avaliacao) };

    // Avaliação anterior opcional, para comparação de evolução.
    if (input.previous_assessment_id) {
      const { data: anterior } = await admin
        .from('physical_assessments')
        .select(ASSESSMENT_COLUMNS)
        .eq('id', input.previous_assessment_id)
        .maybeSingle();
      if (anterior && anterior.tenant_id === tenantId) {
        dados.avaliacao_anterior = formatarAvaliacao(anterior);
      }
    }

    const sinceIso = new Date(Date.now() - RATE_LIMIT_WINDOW_MS).toISOString();
    const { count } = await admin.from('ai_usage_logs').select('id', { count: 'exact', head: true })
      .eq('tenant_id', tenantId).eq('user_id', userId).gte('created_at', sinceIso);
    if ((count ?? 0) >= RATE_LIMIT_MAX) throw rateLimited();

    const { data: gen, error: genErr } = await admin
      .from('ai_generations')
      .insert({ tenant_id: tenantId, created_by: userId, job_type: JOB_TYPE, status: 'pending', input })
      .select('id').single();
    if (genErr || !gen) throw aiFailure('Falha ao registrar a geração de IA.');
    const generationId = gen.id as string;

    const startedAt = Date.now();
    try {
      const result = await callAnthropic(SYSTEM_PROMPT, buildUserMessage(JSON.stringify(dados)));
      const structured = extractStructuredOutput(result.response);
      const parsedOutput = analyzeAssessmentOutputSchema.safeParse(structured);
      if (!parsedOutput.success) {
        console.error('[analyze-assessment] Saída não passou no Zod:', JSON.stringify(structured)?.slice(0, 500));
        throw aiFailure();
      }

      await admin.from('ai_usage_logs').insert({
        tenant_id: tenantId, user_id: userId, job_type: JOB_TYPE, model: result.model,
        input_tokens: result.inputTokens, output_tokens: result.outputTokens,
        estimated_cost_usd: estimateCost(result.model, result.inputTokens, result.outputTokens),
        status: 'succeeded', latency_ms: Date.now() - startedAt,
      });
      await admin.from('ai_generations').update({ status: 'succeeded', output: parsedOutput.data }).eq('id', generationId);
      return jsonResponse({ generation_id: generationId, status: 'succeeded', draft: parsedOutput.data });
    } catch (aiErr) {
      await admin.from('ai_usage_logs').insert({
        tenant_id: tenantId, user_id: userId, job_type: JOB_TYPE, model: null,
        input_tokens: null, output_tokens: null, estimated_cost_usd: null,
        status: 'failed', latency_ms: Date.now() - startedAt,
      });
      await admin.from('ai_generations').update({ status: 'failed', output: null }).eq('id', generationId);
      throw aiErr;
    }
  } catch (err) {
    const info = err instanceof AiError ? `${err.code} ${err.status} ${err.message}` : String(err);
    console.error('[analyze-assessment] Falha:', info);
    return errorResponse(err);
  }
});
